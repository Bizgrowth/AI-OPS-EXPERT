<script>
    (function() {
        const urlParams = new URLSearchParams(window.location.search);
        const returnUrl = urlParams.get('returnUrl');
        const backButton = document.getElementById('back-to-demo-link');

        if (returnUrl) {
            backButton.href = decodeURIComponent(returnUrl);
        } else {
            backButton.href = 'https://aiopsexpert.com#ai-portfolio';
        }

        backButton.addEventListener('click', function(e) {
            if (window.opener) {
                e.preventDefault();
                window.close();
                if (!window.closed) {
                    window.location.href = this.href;
                }
            }
        });
    })();
</script>
